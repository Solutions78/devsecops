md5Check() { command -v md5sum >/dev/null || return 1; [[ -f $2 ]] || return 1; [[ "$(md5sum "$2" | awk '{print $1}')" == "$1" ]]; }

zipf() { zip -r "$1.zip" "$1"; }

copyfile() { [[ -f "$1" ]] && pbcopy <"$1"; }

buf() { cp -a "$1" "$1_$(date +%Y%m%d_%H%M%S)"; }

extract() {
  local verbose=''
  while getopts "v" opt; do [[ "$opt" == "v" ]] && verbose='v'; done
  shift $((OPTIND - 1))
  for f; do
    [[ -f "$f" ]] || { echo "Invalid file: $f"; continue; }
    case "$f" in
      *.tar.bz2|*.tbz|*.tbz2) tar x${verbose}jf "$f" ;;
      *.tar.gz|*.tgz)         tar x${verbose}zf "$f" ;;
      *.tar.xz|*.txz)         tar x${verbose}Jf "$f" ;;
      *.zip|*.war|*.jar)      unzip "$f" ;;
      *.rar)                  unrar x "$f" ;;
      *.7z)                   7za x "$f" ;;
      *.gz)                   gunzip "$f" ;;
      *.bz2)                  bunzip2 "$f" ;;
      *.xz)                   xz --decompress "$f" ;;
      *.Z)                    uncompress "$f" ;;
      *) echo "Cannot extract $f" ;;
    esac
  done
}

chgext() { for f in *."$1"; do mv "$f" "${f%.$1}.$2"; done; }

j2y() { python -c 'import sys,yaml,json; yaml.safe_dump(json.load(sys.stdin), sys.stdout)' <"$1"; }
y2j() { python -c 'import sys,yaml,json; json.dump(yaml.load(sys.stdin, Loader=yaml.FullLoader), sys.stdout, indent=2)' <"$1"; }

gurl() {
  local url="$(git remote get-url origin 2>/dev/null | sed -e 's/^git@/https:\/\//' -e 's/:/\//' -e 's/\.git$//')"
  [[ -n "$url" ]] && echo "$url"
}

applyignore() { git ls-files -ci --exclude-standard -z | xargs -0 git rm --cached; }

rollback() {
  [[ -z "$1" ]] && echo "Commit hash required" && return 1
  git rev-parse --quiet --verify "$1" >/dev/null || { echo "Invalid commit"; return 1; }
  read -q "REPLY?Rollback to $1 (Y/N)? " && echo
  [[ $REPLY =~ [Yy] ]] && git reset --hard "$1"
}
